# Backend runtime dependencies. Kept in sync with pyproject.toml so pip
# users get the same versions as the editable install.
#
# Install:
#   python -m venv .venv
#   .\.venv\Scripts\python.exe -m pip install --upgrade pip
#   .\.venv\Scripts\python.exe -m pip install -r requirements.txt
#
# Tested on Python 3.14 (Windows). The opencv-python-headless build is
# preferred over opencv-python — no GUI is needed (everything renders in
# the browser) and it sidesteps GTK/Qt dependency chains.

fastapi>=0.115
uvicorn[standard]>=0.32
pydantic>=2.9
pydantic-settings>=2.6
python-multipart>=0.0.12

opencv-python-headless>=4.10
numpy>=1.26
scipy>=1.13

# URL video ingest — pulls videos directly from most video sites so
# they don't have to be manually downloaded first.
yt-dlp>=2024.10

# Optional: SOTA neural beat tracker (CPJKU BEAT This, 2024) for audio
# mode. Pulls in PyTorch + torchaudio (~250 MB CPU-only) plus a ~30 MB
# model auto-downloaded on first use. Job falls back to spectral-flux
# detection if beat_this isn't installed, so this is genuinely optional.
beat_this>=1.1

# Optional: YOLOv8-pose body-keypoint tracker for Pose detection mode.
# Reuses the PyTorch install that beat_this already pulls in. The
# yolov8n-pose.pt weights (~6.5 MB) auto-download on first job to
# torch's hub cache. Pose mode is only available with this installed.
ultralytics>=8.2

# Optional: SAM 2 (Meta 2024) for Object detection mode. Click any
# object in one frame; the model tracks it through the whole video.
# The sam2.1_hiera_tiny.pt checkpoint (~150 MB) auto-downloads to
# torch's hub cache on first job. Reuses the existing PyTorch install.
# decord is a transitive dep sam2's own pyproject forgets to declare;
# without it SAM 2's init_state raises ModuleNotFoundError on MP4s.
sam2>=1.1
decord>=0.6

# Desktop-app launcher (start-desktop.bat / backend/desktop.py). Only
# needed if you want a native window instead of the browser tab.
# Windows uses WebView2 (bundled with Windows 10 21H1+ / Windows 11).
pywebview>=5.3
